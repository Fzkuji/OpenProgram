# OpenProgram dependencies.
#
# pyproject.toml is the SINGLE source of truth. This file used to mirror
# [project.dependencies] by hand, which meant two lists to keep in sync and
# one of them silently drifting. It now just installs the package.
#
#     pip install -r requirements.txt     # editable install of this repo
#
# equivalent to, and interchangeable with:
#
#     pip install -e .
#
# To add or change a dependency, edit [project.dependencies] in
# pyproject.toml. Do not add bare package lines here.
#
# Optional extras (not installed by the line below — add them explicitly):
#     pip install -e ".[channels]"   # discord / slack / wechat QR
#     pip install -e ".[browser]"    # playwright (then: playwright install chromium)
#     pip install -e ".[dev]"        # pytest / ruff
#     pip install -e ".[all]"        # channels + browser
#
# Agentic programs (gui / research / wiki harnesses) are NOT installed from
# here. Each is a self-describing repo pulled in by:
#     openprogram programs install <name>
